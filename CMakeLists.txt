cmake_minimum_required(VERSION 3.24)
project(FlowTunnel LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "FlowTunnel requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(flowtunnel_core src/protocol.cpp src/packet.cpp src/divert.cpp src/stream.cpp)
target_include_directories(flowtunnel_core PUBLIC src third_party/asio-asio-1-36-0/asio/include third_party/WinDivert-2.2.2-A/include)
target_compile_definitions(flowtunnel_core PUBLIC ASIO_STANDALONE ASIO_NO_DEPRECATED WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(flowtunnel_core PUBLIC bcrypt ws2_32 mswsock)
target_compile_options(flowtunnel_core PRIVATE /W4 /permissive- /utf-8)
add_executable(flowtunnel src/main.cpp src/service.cpp src/tunnel.cpp)
target_include_directories(flowtunnel PRIVATE third_party/tomlplusplus-3.4.0/include)
target_link_libraries(flowtunnel PRIVATE flowtunnel_core crypt32 advapi32)
target_compile_options(flowtunnel PRIVATE /W4 /permissive- /utf-8)
set_target_properties(flowtunnel PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(flowtunnel PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/bin")
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(flowtunnel_tests tests/native_test.cpp tests/protocol_test.cpp tests/packet_test.cpp tests/stream_test.cpp tests/datagram_ack_test.cpp)
    target_link_libraries(flowtunnel_tests PRIVATE flowtunnel_core)
    target_compile_options(flowtunnel_tests PRIVATE /W4 /permissive- /utf-8)
    add_test(NAME native COMMAND flowtunnel_tests)
endif()
